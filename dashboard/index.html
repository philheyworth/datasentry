Scanner Deployment Snippet

Pass these environment variables when deploying the scanner to a customer machine. Use --customer-id and --customer-name to group machines by customer in the dashboard.

# Windows — set these then run the .exe
set DATASENTRY_API_URL=https://your-api.example.com
set DATASENTRY_API_KEY=dsk_your_key_here
set DATASENTRY_CUSTOMER_ID=acme-corp
set DATASENTRY_CUSTOMER_NAME=Acme Corporation
datasentry-scanner.exe

# macOS / Linux
DATASENTRY_API_URL=https://your-api.example.com \
DATASENTRY_API_KEY=dsk_your_key_here \
DATASENTRY_CUSTOMER_ID=acme-corp \
DATASENTRY_CUSTOMER_NAME="Acme Corporation" \
./datasentry-scanner --cli
